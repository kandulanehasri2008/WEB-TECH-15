/* =====================================================
   Meridian Diagnostics – Toggle switch (script.js)
   Setting: "Home sample collection"
   ON  -> address box appears and ₹150 is added to the total
   OFF -> address box is hidden and the charge is removed
   The choice is saved in Local Storage.

   Uses: addEventListener, checked, classList.toggle,
         localStorage.setItem, localStorage.getItem
   ===================================================== */

const STORAGE_KEY = "meridian_home_collection";
const HOME_CHARGE = 150;

const toggle      = document.getElementById("home-toggle");
const stateText   = document.getElementById("home-state");
const card        = document.getElementById("home-card");
const description = document.getElementById("home-desc");
const homeField   = document.getElementById("home-field");
const homeRow     = document.getElementById("row-home");
const testSelect  = document.getElementById("test");
const feeTest     = document.getElementById("fee-test");
const feeTotal    = document.getElementById("fee-total");

function formatRupees(amount) {
  return "₹" + amount.toLocaleString("en-IN");
}

/* Update the price summary */
function updateTotal() {
  const testFee = Number(testSelect.value);
  const total = testFee + (toggle.checked ? HOME_CHARGE : 0);
  feeTest.textContent = formatRupees(testFee);
  feeTotal.textContent = formatRupees(total);
}

/* Update the whole page to match the switch */
function applySetting(isOn) {
  toggle.checked = isOn;
  stateText.textContent = isOn ? "On" : "Off";
  description.textContent = isOn
    ? "On: a phlebotomist will visit you. A ₹150 collection charge applies."
    : "Off: you will visit the centre.";

  card.classList.toggle("is-on", isOn);     /* highlight the card */
  homeField.hidden = !isOn;                 /* show / hide the address box */
  homeRow.hidden = !isOn;                   /* show / hide the extra charge row */
  updateTotal();
}

/* Read the saved setting when the page opens */
let saved = false;
try {
  saved = localStorage.getItem(STORAGE_KEY) === "true";
} catch (error) {
  saved = false;                            /* storage blocked: start with Off */
}
applySetting(saved);

/* Save the new setting whenever the switch is used */
toggle.addEventListener("change", function () {
  applySetting(toggle.checked);
  try {
    localStorage.setItem(STORAGE_KEY, String(toggle.checked));
  } catch (error) {
    /* storage unavailable: the switch still works for this visit */
  }
});

/* A different test changes the total */
testSelect.addEventListener("change", updateTotal);
